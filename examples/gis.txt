CREATE GIS water_supply USING 4326;
USE GIS water_supply;

CREATE ENTITY River (
  id Long IDENTIFIER DISPLAY_STRING,
  name String REQUIRED,
  length Long,
  geometry MultiLineString
);

CREATE ENTITY Sea (
  id Long IDENTIFIER DISPLAY_STRING,
  name String REQUIRED,
  area Long,
  geometry MultiPolygon
);

CREATE ENTITY MaritimeBuoy (
  id Long IDENTIFIER DISPLAY_STRING,
  name String REQUIRED,
  location String,
  geometry Point
);

CREATE TILE LAYER osm AS 'OpenStreetMap' (
  url 'https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png'
);

CREATE WMS STYLE river_style (
    geometryType LINESTRING,
    strokeColor #0066cc
);

CREATE WMS LAYER river_layer AS 'Rivers' (
  River river_style
);

CREATE WMS STYLE sea_style (
    geometryType POLYGON,
    fillColor #0099ff,
    strokeColor #0066cc,
    fillOpacity 0.5,
    strokeOpacity 1.0
);

CREATE WMS LAYER sea_layer AS 'Seas' (
  Sea sea_style
);

CREATE WMS STYLE buoy_style (
    geometryType POINT,
    fillColor #ff0000,
    strokeColor #ff0000,
    fillOpacity 1.0,
    strokeOpacity 1.0
);

CREATE WMS LAYER buoy_layer AS 'Maritime Buoys' (
  MaritimeBuoy buoy_style
);

CREATE MAP main AS 'Water Supply Map' (
  osm IS_BASE_LAYER,
  river_layer,
  sea_layer,
  buoy_layer
);

GENERATE GIS water_supply;
