CREATE GIS water_supply_marine USING 4326; USE GIS water_supply_marine; CREATE ENTITY River ( id Long IDENTIFIER DISPLAY_STRING, name String REQUIRED, river_code String REQUIRED UNIQUE, length Double, geometry LineString ); CREATE ENTITY Sea ( id Long IDENTIFIER DISPLAY_STRING, name String REQUIRED, water_type String, area Double, geometry Polygon ); CREATE ENTITY MaritimeBuoy ( id Long IDENTIFIER DISPLAY_STRING, buoy_id String REQUIRED UNIQUE, name String, buoy_type String, status String, geometry Point ); CREATE TILE LAYER osm AS 'OpenStreetMap' ( url 'https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png' ); CREATE WMS STYLE river_style ( geometryType LINESTRING, strokeColor #0066cc, strokeOpacity 1.0, strokeWidth 2 ); CREATE WMS LAYER river_layer AS 'Rivers' ( River river_style ); CREATE WMS STYLE sea_style ( geometryType POLYGON, fillColor #3399ff, strokeColor #0066cc, fillOpacity 0.4, strokeOpacity 0.8 ); CREATE WMS LAYER sea_layer AS 'Seas' ( Sea sea_style ); CREATE WMS STYLE buoy_style ( geometryType POINT, fillColor #ff6600, strokeColor #cc3300, fillOpacity 0.9, strokeOpacity 1.0 ); CREATE WMS LAYER buoy_layer AS 'Maritime Buoys' ( MaritimeBuoy buoy_style ); CREATE MAP main AS 'Water Supply Marine Map' ( osm IS_BASE_LAYER, river_layer, sea_layer, buoy_layer ); GENERATE GIS water_supply_marine;
